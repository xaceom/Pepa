### 4. Создать базовый плагин (30 минут)

**src/main.ts:**
```typescript
import { Plugin, Notice } from 'obsidian';
import {
  TelegramEmojiSettings,
  DEFAULT_SETTINGS,
  TelegramEmojiSettingTab
} from './ui/SettingsTab';

export default class TelegramEmojiPlugin extends Plugin {
  settings: TelegramEmojiSettings;

  async onload() {
    console.log('Loading Telegram Emoji plugin');

    // Загрузить настройки
    await this.loadSettings();

    // Добавить ribbon icon
    this.addRibbonIcon('smile', 'Open Telegram Emoji', () => {
      new Notice('Telegram Emoji picker (coming soon)');
    });

    // Добавить command
    this.addCommand({
      id: 'open-emoji-picker',
      name: 'Open Emoji Picker',
      callback: () => {
        new Notice('Opening emoji picker...');
      },
      hotkeys: [{ modifiers: ['Ctrl'], key: 'e' }]
    });

    // Добавить settings tab
    this.addSettingTab(new TelegramEmojiSettingTab(this.app, this));
  }

  onunload() {
    console.log('Unloading Telegram Emoji plugin');
  }

  async loadSettings() {
    this.settings = Object.assign({}, DEFAULT_SETTINGS, await this.loadData());
  }

  async saveSettings() {
    await this.saveData(this.settings);
  }

  async clearCache() {
    // TODO: Implement cache clearing
    new Notice('Cache cleared!');
  }
}
```

### 6. Первый тест API (15 минут)

**tests/manual/test-api.ts:**
```typescript
import { TelegramBotAPI } from '../../src/api/TelegramBotAPI';

async function testAPI() {
  const token = process.env.TELEGRAM_BOT_TOKEN;
  if (!token) {
    throw new Error('TELEGRAM_BOT_TOKEN not set');
  }

  const api = new TelegramBotAPI(token);

  try {
    // Тестовый emoji ID (замените на реальный)
    const testEmojiId = '5368324170671202286';

    console.log('Fetching emoji info...');
    const stickers = await api.getCustomEmojiStickers([testEmojiId]);

    console.log('Success!', stickers);

    // Попробовать скачать файл
    if (stickers.length > 0) {
      console.log('Downloading emoji file...');
      const fileData = await api.downloadFile(stickers[0].file_id);
      console.log(`Downloaded ${fileData.byteLength} bytes`);
    }

  } catch (error) {
    console.error('API test failed:', error);
  }
}

testAPI();
```

## 🆘 Если что-то пошло не так

### Ошибка компиляции TypeScript
```bash
# Переустановить зависимости
rm -rf node_modules package-lock.json
npm install
```

### Плагин не загружается в Obsidian
1. Проверьте console (Ctrl+Shift+I)
2. Убедитесь что manifest.json валиден
3. Перезапустите Obsidian

### API тест не работает
1. Проверьте, что Bot Token правильный
2. Проверьте интернет соединение
3. Попробуйте curl тест из DEVELOPMENT_PLAN.md
